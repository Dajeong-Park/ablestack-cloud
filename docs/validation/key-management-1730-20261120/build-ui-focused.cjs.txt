const os = require('os')
const cpus = os.cpus
os.cpus = () => cpus().slice(0, 2)
const Service = require('/root/work/ablestack-cloud/ui/node_modules/@vue/cli-service/lib/Service')
const service = new Service('/root/work/ablestack-cloud/ui')
service.init('production')
service.webpackChainFns.push(config => {
  config.optimization.minimize(false)
  config.devtool(false)
})
service.run('build', { mode: 'production', dest: '/root/work/ablestack-cloud/ui/dist' }).catch(error => {
  process.stderr.write(error.stack + '\n')
  process.exitCode = 1
})
