def run(args):
    return subprocess.run(args, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True, timeout=20)

def lsblk():
    result = run(["lsblk", "-b", "-J", "-o", "NAME,PATH,TYPE,FSTYPE,UUID,SERIAL,SIZE,MOUNTPOINT"])
    if result.returncode != 0:
        raise SystemExit(result.stderr.strip() or "lsblk failed")
    return json.loads(result.stdout).get("blockdevices", [])

def walk(devices, parent_disk_path=None, parent_path=None):
    for dev in devices:
        dev["_parentDiskPath"] = dev.get("path") if dev.get("type") == "disk" else parent_disk_path
        dev["_parentPath"] = parent_path
        yield dev
        for child in dev.get("children") or []:
            yield from walk([child], dev.get("_parentDiskPath"), dev.get("path"))

def compact(value):
    return "".join(ch for ch in str(value or "").lower() if ch.isalnum())

def token_matches(value, tokens):
    candidate = compact(value)
    if not candidate:
        return False
    for token in tokens:
        if len(token) >= 8 and (token in candidate or candidate in token):
            return True
    return False

def parse_size(value):
    try:
        return int(value)
    except (TypeError, ValueError):
        return 0

def size_matches(dev, expected_size):
    if expected_size <= 0:
        return False
    actual_size = parse_size(dev.get("size"))
    if actual_size <= 0:
        return False
    tolerance = max(1024 * 1024, expected_size // 100)
    return abs(actual_size - expected_size) <= tolerance

def is_root_disk_tree(dev):
    return dev.get("_parentDiskPath") in root_disk_paths

def has_child_filesystem_or_mount(dev):
    path = dev.get("path")
    if not path:
        return False
    for child in devices:
        if child.get("_parentDiskPath") == path and child.get("path") != path and (child.get("fstype") or child.get("mountpoint")):
            return True
    return False

def device_matches_identity(dev):
    if not identity_tokens:
        return False
    return any(token_matches(dev.get(key), identity_tokens) for key in ("name", "path", "serial", "uuid"))

def safe_candidate(dev, allow_mounted_match=False):
    if not dev or not dev.get("path"):
        return False
    if dev.get("type") not in ("disk", "part"):
        return False
    if is_root_disk_tree(dev):
        return False
    if dev.get("mountpoint") and not (allow_mounted_match and device_matches_identity(dev)):
        return False
    if import_mode in format_modes:
        return dev.get("type") == "disk" and not dev.get("fstype") and not dev.get("mountpoint") and not has_child_filesystem_or_mount(dev)
    return True

def select_single(candidates, reason):
    unique = []
    seen = set()
    for dev in candidates:
        path = dev.get("path")
        if path and path not in seen:
            unique.append(dev)
            seen.add(path)
    if len(unique) == 1:
        return unique[0].get("path")
    if len(unique) > 1:
        paths = ", ".join(dev.get("path") for dev in unique)
        raise SystemExit(f"Ambiguous Storage Service volume device candidates for {reason}: {paths}")
    return None

def fstab_escape(value):
    return str(value).replace("\\", "\\\\").replace(" ", "\\040").replace("\t", "\\011")

def update_fstab(marker, entry=None):
    fstab_path = "/etc/fstab"
    lines = []
    if os.path.exists(fstab_path):
        with open(fstab_path, "r", encoding="utf-8") as fstab_fp:
            for line in fstab_fp:
                if marker not in line:
                    lines.append(line)
    if entry:
        lines.append(entry.rstrip("\n") + "\n")
    directory = os.path.dirname(fstab_path)
    fd, tmp_path = tempfile.mkstemp(prefix=".fstab.", dir=directory, text=True)
    try:
        with os.fdopen(fd, "w", encoding="utf-8") as tmp_fp:
            tmp_fp.writelines(lines)
        os.replace(tmp_path, fstab_path)
    finally:
        if os.path.exists(tmp_path):
            os.unlink(tmp_path)

def daemon_reload():
    if shutil.which("systemctl"):
        subprocess.run(["systemctl", "daemon-reload"], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, check=False, timeout=20)

def blkid_value(path, field):
    result = run(["blkid", "-s", field, "-o", "value", path])
    if result.returncode == 0 and result.stdout.strip():
        return result.stdout.strip().splitlines()[0]
    return ""

def persist_backing_mount(device, target_path, fs):
    fs_uuid = blkid_value(device, "UUID")
    if not fs_uuid:
        raise SystemExit(f"Unable to resolve filesystem UUID for Storage Service backing volume {device}")
    fstype = fs or blkid_value(device, "TYPE") or "auto"
    options = "defaults,nofail,x-systemd.device-timeout=30"
    entry = f"UUID={fstab_escape(fs_uuid)} {fstab_escape(target_path)} {fstab_escape(fstype)} {options} 0 2 # {managed_marker}"
    update_fstab(managed_marker, entry)
    daemon_reload()
    return fs_uuid

def format_empty_device():
    fs = str(requested_filesystem or "xfs").lower()
    allowed = {"xfs", "ext4"}
    if fs not in allowed:
        raise SystemExit(f"Unsupported filesystem for empty backing volume format: {fs}")
    tool = f"mkfs.{fs}"
    if not shutil.which(tool):
        raise SystemExit(f"Missing filesystem formatter: {tool}")
    result = run([tool, "-f", device] if fs == "xfs" else [tool, "-F", device])
    if result.returncode != 0:
        raise SystemExit(result.stderr.strip() or f"failed to format {device} as {fs}")
    return fs

def resolve_backing_path(volume_mount_path, relative_path, create_missing):
    if not relative_path:
        return volume_mount_path
    cleaned = relative_path.strip().strip("/")
    if not cleaned or cleaned == "." or cleaned.startswith("..") or "/../" in f"/{cleaned}/":
        raise SystemExit(f"Invalid backing volume relative share path: {relative_path}")
    if cleaned == "export":
        raise SystemExit("The legacy /export root cannot be used as a share backing directory")
    if cleaned.startswith("export/") and cleaned.count("/") != 1:
        raise SystemExit(f"Storage Service share path must be a direct child of /export: /{cleaned}")
    root = os.path.realpath(volume_mount_path)
    target = os.path.realpath(os.path.join(volume_mount_path, cleaned))
    if target != root and not target.startswith(root + os.sep):
        raise SystemExit(f"Backing volume relative share path escapes the mounted volume: {relative_path}")
    if os.path.exists(target) and not os.path.isdir(target):
        raise SystemExit(f"Backing volume relative share path exists and is not a directory: {relative_path}")
    if not os.path.exists(target):
        if not create_missing:
            raise SystemExit(f"Backing volume relative share path does not exist: {relative_path}")
        os.makedirs(target, exist_ok=True)
    return target
